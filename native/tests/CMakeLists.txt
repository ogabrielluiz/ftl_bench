cmake_minimum_required(VERSION 3.16)
project(ftl_bench_economy_tests LANGUAGES CXX)
add_executable(economy_test economy_test.cpp)
target_compile_features(economy_test PRIVATE cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(economy_test PRIVATE -Wall -Wextra -pedantic)
endif()
enable_testing()
add_test(NAME native_economy COMMAND economy_test)

add_executable(routing_test routing_test.cpp)
target_compile_features(routing_test PRIVATE cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(routing_test PRIVATE -Wall -Wextra -pedantic)
endif()
add_test(NAME native_sector_routing COMMAND routing_test)

add_executable(fresh_start_test fresh_start_test.cpp)
target_compile_features(fresh_start_test PRIVATE cxx_std_17)
target_include_directories(fresh_start_test PRIVATE stubs ../include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(fresh_start_test PRIVATE -Wall -Wextra -pedantic)
endif()
add_test(NAME native_fresh_start COMMAND fresh_start_test)
